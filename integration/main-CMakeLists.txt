# add GATT DB creation
message("\n===")
if ( DEFINED GATT_PATH )
    message("${BUILD_MESSAGE} with GATT DB")
    # compile_gatt.py must be run through Python explicitly: on the Windows
    # setup this was developed on, executing it directly launched Node.js.
    # Use the interpreter ESP-IDF itself uses.
    idf_build_get_property(GATT_PYTHON PYTHON)
    if( NOT GATT_PYTHON )
        set(GATT_PYTHON "C:/Espressif/tools/python/v6.1/venv/Scripts/python.exe")
    endif()

    set(GATT_COMPILER
        "${BTSTACK_ROOT}/tool/compile_gatt.py")

    # compile_gatt.py output is named after the .gatt file, not the example.
    get_filename_component(GATT_NAME "${GATT_PATH}" NAME_WLE)
    set(GATT_HEADER
        "${CMAKE_CURRENT_BINARY_DIR}/${GATT_NAME}.h")

    add_custom_command(
        OUTPUT "${GATT_HEADER}"
        COMMAND "${GATT_PYTHON}"
                "${GATT_COMPILER}"
                "${GATT_PATH}"
                "${GATT_HEADER}"
        DEPENDS "${GATT_PATH}" "${GATT_COMPILER}"
        WORKING_DIRECTORY "${BTSTACK_ROOT}/tool"
        COMMENT "Generating ${GATT_NAME}.h using Python"
        VERBATIM
    )

    target_sources(${COMPONENT_LIB} PRIVATE "${GATT_HEADER}")

    target_include_directories(${COMPONENT_LIB}
        PRIVATE "${CMAKE_CURRENT_BINARY_DIR}")
    if( EXAMPLE STREQUAL "wmr_bridge" )
        target_include_directories(${COMPONENT_LIB}
            PRIVATE "${WMR_BRIDGE_DIR}/src")
    endif()
else()
    message(${BUILD_MESSAGE})
endif()
message("===\n")
